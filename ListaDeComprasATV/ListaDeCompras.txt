import { useState } from 'react';
import {
  FlatList,
  Keyboard,
  StyleSheet,
  Text,
  TextInput,
  TouchableOpacity,
  View,
} from 'react-native';

// tipo de cada item da lista
type Item = {
  id: string;
  nome: string;
  comprado: boolean;
};

// props do componente, só precisa do nome da pessoa mesmo
type Props = {
  nome: string;
};

export default function ListaCompras({ nome }: Props) {
  const [itens, setItens] = useState<Item[]>([]); // lista de itens
  const [pesquisa, setPesquisa] = useState(''); // texto digitado na busca
  const [novoItem, setNovoItem] = useState(''); // texto do input de adicionar
  const [mostrarAdicionar, setMostrarAdicionar] = useState(false); // mostra ou esconde a caixinha de add
  const [menuAberto, setMenuAberto] = useState<string | null>(null); // guarda o id do item com o menuzinho aberto

  // funcao que roda quando clica em "Adicionar"
  function adicionarItem() {
    // se o campo tiver vazio nao deixa adicionar
    if (novoItem.trim() == '') {
      return;
    }

    // monta o item novo
    const item: Item = {
      id: Date.now().toString(), // usei a data como id, meio gambiarra mas resolve
      nome: novoItem,
      comprado: false,
    };

    setItens([...itens, item]); // junta com os itens que ja tinham
    setNovoItem(''); // limpa o campo depois de adicionar
    setMostrarAdicionar(false); // fecha a caixinha de novo

    Keyboard.dismiss(); // fecha o teclado do celular
  }

  // marca ou desmarca o item quando clica em cima dele
  function marcarItem(id: string) {
    const novaLista = itens.map((item) => {
      if (item.id == id) {
        return { ...item, comprado: !item.comprado }; // inverte o valor
      }
      return item;
    });

    setItens(novaLista);
    setMenuAberto(null); // fecha o menu se tiver algum aberto
  }

  // remove o item da lista
  function excluirItem(id: string) {
    const novaLista = itens.filter((item) => item.id != id);
    setItens(novaLista);
    setMenuAberto(null);
  }

  // filtra os itens de acordo com o texto da pesquisa
  const itensFiltrados = itens.filter((item) => {
    return item.nome.toLowerCase().indexOf(pesquisa.toLowerCase()) !== -1;
  });

  // conta quantos itens ainda faltam comprar
  let pendentes = 0;
  for (let i = 0; i < itens.length; i++) {
    if (!itens[i].comprado) {
      pendentes = pendentes + 1;
    }
  }

  return (
    <View style={styles.container}>
      {/* nome da pessoa la em cima */}
      <Text style={styles.titulo}>Olá, {nome}!</Text>
      <Text style={styles.subtitulo}>Minha lista de compras</Text>

      {/* linha com o campo de busca e o botaozinho de + */}
      <View style={styles.pesquisaLinha}>
        <TextInput
          style={styles.pesquisa}
          placeholder="Pesquisar item..."
          placeholderTextColor="#999"
          value={pesquisa}
          onChangeText={setPesquisa}
        />

        <TouchableOpacity
          style={styles.botaoMais}
          onPress={() => {
            setMostrarAdicionar(!mostrarAdicionar);
            setMenuAberto(null);
          }}
        >
          <Text style={styles.mais}>+</Text>
        </TouchableOpacity>
      </View>

      {/* essa caixa so aparece quando clica no + */}
      {mostrarAdicionar && (
        <View style={styles.caixaAdicionar}>
          <TextInput
            style={styles.inputAdicionar}
            placeholder="Nome do produto"
            placeholderTextColor="#999"
            value={novoItem}
            onChangeText={setNovoItem}
            autoFocus
            onSubmitEditing={adicionarItem}
          />

          <TouchableOpacity style={styles.botaoAdicionar} onPress={adicionarItem}>
            <Text style={styles.textoBotao}>Adicionar</Text>
          </TouchableOpacity>
        </View>
      )}

      {/* lista com os itens ja filtrados pela pesquisa */}
      <FlatList
        data={itensFiltrados}
        keyExtractor={(item) => item.id}
        contentContainerStyle={styles.lista}
        showsVerticalScrollIndicator={false}
        ListEmptyComponent={
          // mensagem quando nao tem nada pra mostrar
          <Text style={styles.listaVazia}>
            {pesquisa ? 'Nenhum item encontrado.' : 'Sua lista está vazia.'}
          </Text>
        }
        renderItem={({ item }) => (
          <View style={styles.itemContainer}>
            <TouchableOpacity style={styles.item} onPress={() => marcarItem(item.id)}>
              {/* bolinha do check, fica preta quando marca o item */}
              <View style={[styles.checkbox, item.comprado && styles.checkboxMarcado]}>
                {item.comprado && <Text style={styles.check}>✓</Text>}
              </View>

              {/* risca o nome quando o item ja foi comprado */}
              <Text style={[styles.nomeItem, item.comprado && styles.comprado]}>
                {item.nome}
              </Text>
            </TouchableOpacity>

            {/* botao dos 3 pontinhos, abre o menu de opcoes */}
            <TouchableOpacity
              style={styles.botaoMenu}
              onPress={() => {
                if (menuAberto == item.id) {
                  setMenuAberto(null);
                } else {
                  setMenuAberto(item.id);
                }
              }}
            >
              <Text style={styles.pontos}>⋮</Text>
            </TouchableOpacity>

            {/* menu simples, por enquanto so tem excluir */}
            {menuAberto == item.id && (
              <View style={styles.menu}>
                <TouchableOpacity onPress={() => excluirItem(item.id)} style={styles.excluir}>
                  <Text style={styles.textoExcluir}>Excluir item</Text>
                </TouchableOpacity>
              </View>
            )}
          </View>
        )}
      />

      {/* rodape mostrando quantos itens ainda faltam */}
      <View style={styles.rodape}>
        <Text style={styles.pendentes}>Itens pendentes: {pendentes}</Text>
      </View>
    </View>
  );
}
