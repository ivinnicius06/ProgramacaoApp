import React from 'react';
import {
  View,
  Text,
  TextInput,
  TouchableOpacity,
  Image,
  StyleSheet,
  KeyboardAvoidingView,
  Platform,
} from 'react-native';

// props que a tela de login recebe
type Props = {
  nome: string;
  onMudarNome: (texto: string) => void;
  onEntrar: () => void;
};

export default function TelaLogin({ nome, onMudarNome, onEntrar }: Props) {
  // no ios usa "padding" e no android usa "height", senao o teclado cobre o campo
  const comportamentoTeclado = Platform.OS === 'ios' ? 'padding' : 'height';

  // desabilita o botao enquanto o campo de nome estiver vazio
  const botaoDesabilitado = nome.trim() == '';

  return (
    <KeyboardAvoidingView style={styles.tela} behavior={comportamentoTeclado}>
      <View style={styles.conteudo}>
        {/* icone do carrinho la em cima */}
        <Image
          source={require('../assets/carrinho.png')}
          style={styles.icone}
          resizeMode="contain"
        />

        <Text style={styles.titulo}>Lista de Compras</Text>

        <Text style={styles.subtitulo}>
          Organize suas compras com facilidade
        </Text>

        {/* campo pra digitar o nome */}
        <TextInput
          style={styles.campo}
          placeholder="Insira seu nome"
          placeholderTextColor="#999"
          value={nome}
          onChangeText={onMudarNome}
          autoCapitalize="words"
          returnKeyType="done"
          onSubmitEditing={onEntrar}
        />

        {/* botao de entrar, fica cinza e "desligado" quando nao tem nome */}
        <TouchableOpacity
          style={[styles.botao, botaoDesabilitado && styles.botaoDesabilitado]}
          onPress={onEntrar}
        >
          <Text style={styles.textoBotao}>Entrar</Text>
        </TouchableOpacity>
      </View>
    </KeyboardAvoidingView>
  );
}

// estilos da tela toda
const styles = StyleSheet.create({
  tela: {
    flex: 1,
    backgroundColor: '#fff',
  },
  conteudo: {
    flex: 1,
    alignItems: 'center',
    justifyContent: 'center',
    paddingHorizontal: 32,
  },
  icone: {
    width: 110,
    height: 110,
    marginBottom: 24,
  },
  titulo: {
    fontSize: 28,
    fontWeight: '700',
    color: '#1a1a1a',
    marginBottom: 6,
  },
  subtitulo: {
    fontSize: 14,
    color: '#888',
    marginBottom: 48,
    textAlign: 'center',
  },
  campo: {
    width: '100%',
    height: 52,
    borderWidth: 1.5,
    borderColor: '#e0e0e0',
    borderRadius: 12,
    paddingHorizontal: 16,
    fontSize: 16,
    color: '#1a1a1a',
    backgroundColor: '#fafafa',
    marginBottom: 16,
  },
  botao: {
    width: '100%',
    height: 52,
    backgroundColor: '#1a1a1a',
    borderRadius: 12,
    alignItems: 'center',
    justifyContent: 'center',
  },
  botaoDesabilitado: {
    backgroundColor: '#c0c0c0', // fica cinza quando nao da pra clicar
  },
  textoBotao: {
    color: '#fff',
    fontSize: 16,
    fontWeight: '600',
  },
});
